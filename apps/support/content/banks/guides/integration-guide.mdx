## Phase 2: Authorization API (Week 2-3)

### 2.1 Implement Authorization Endpoint

```javascript
// Receive authorization request from Vinr
app.post('/api/vinr/authorize', async (req, res) => {
  const {
    transaction_id,
    merchant_id,
    amount,
    currency,
    payment_method,
    billing_address
  } = req.body;

  try {
    // Process with your card processor
    const result = await processCardAuthorization({
      amount,
      currency,
      card: payment_method.card,
      billing: billing_address
    });

    // Return response
    res.json({
      authorization_id: generateAuthId(),
      status: result.approved ? 'approved' : 'declined',
      authorization_code: result.authCode,
      decline_reason: result.declineReason,
      processor_response: result.processorCode
    });
  } catch (error) {
    res.status(500).json({
      error: 'processing_error',
      message: error.message
    });
  }
});
```

## Phase 4: Webhooks (Week 5-6)

### 4.1 Set Up Webhook Receiver

```javascript
app.post('/webhooks/vinr', async (req, res) => {
  // Verify signature
  const signature = req.headers['vinr-signature'];
  if (!verifySignature(req.body, signature)) {
    return res.status(401).json({error: 'Invalid signature'});
  }

  // Respond immediately
  res.json({received: true});

  // Process asynchronously
  await processWebhookAsync(req.body);
});

async function processWebhookAsync(event) {
  switch (event.type) {
    case 'merchant.verified':
      await onMerchantVerified(event.data);
      break;
    case 'settlement.initiated':
      await onSettlementInitiated(event.data);
      break;
    case 'chargeback.received':
      await onChargebackReceived(event.data);
      break;
  }
}
```

### 4.2 Implement Signature Verification

```javascript
const crypto = require('crypto');

function verifySignature(payload, signature, secret) {
  const expectedSignature = crypto
    .createHmac('sha256', secret)
    .update(JSON.stringify(payload))
    .digest('hex');
  
  return crypto.timingSafeEqual(
    Buffer.from(signature.replace('sha256=', '')),
    Buffer.from(expectedSignature)
  );
}
```

## Phase 5: Security Implementation (Week 6-7)

### 5.1 Implement IP Whitelisting

```javascript
const VINR_IPS = [
  '52.123.45.67',
  '52.123.45.68',
  '52.123.45.69'
];

app.use((req, res, next) => {
  if (req.path.startsWith('/api/vinr/')) {
    const clientIP = req.ip;
    if (!VINR_IPS.includes(clientIP)) {
      return res.status(403).json({error: 'Forbidden'});
    }
  }
  next();
});
```

### 5.2 Implement Rate Limiting

```javascript
const rateLimit = require('express-rate-limit');

const apiLimiter = rateLimit({
  windowMs: 1000, // 1 second
  max: 100 // 100 requests per second
});

app.use('/api/vinr/', apiLimiter);
```

```javascript
// Implement monitoring
const metrics = {
  authorizationSuccess: new Counter('authorization_success'),
  authorizationFailure: new Counter('authorization_failure'),
  responseTime: new Histogram('response_time')
};

app.post('/api/vinr/authorize', async (req, res) => {
  const start = Date.now();
  
  try {
    const result = await processAuthorization(req.body);
    metrics.authorizationSuccess.inc();
    res.json(result);
  } catch (error) {
    metrics.authorizationFailure.inc();
    res.status(500).json({error: error.message});
  } finally {
    metrics.responseTime.observe(Date.now() - start);
  }
});
```
